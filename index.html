const canvas = document.getElementById("canvas"); const ctx = canvas.getContext("2d"); let w, h; function resize() { w = canvas.width = window.innerWidth; h = canvas.height = window.innerHeight; } window.addEventListener("resize", resize); resize(); let stars = []; const STAR_COUNT = 1200; for (let i = 0; i < STAR_COUNT; i++) { stars.push({ x: (Math.random() - 0.5) * w, y: (Math.random() - 0.5) * h, z: Math.random() * w }); } let speed = 2; let maxSpeed = 30; let flashTriggered = false; function animate() { // MOTION BLUR (trail effect) ctx.fillStyle = "rgba(0,0,0,0.25)"; ctx.fillRect(0, 0, w, h); for (let s of stars) { let prevZ = s.z; s.z -= speed; if (s.z <= 0) { s.x = (Math.random() - 0.5) * w; s.y = (Math.random() - 0.5) * h; s.z = w; } let k = 128 / s.z; let px = s.x * k + w / 2; let py = s.y * k + h / 2; let kPrev = 128 / prevZ; let pxPrev = s.x * kPrev + w / 2; let pyPrev = s.y * kPrev + h / 2; // STAR TRAIL ctx.strokeStyle = "rgba(120,180,255,0.8)"; ctx.lineWidth = (1 - s.z / w) * 3; ctx.beginPath(); ctx.moveTo(pxPrev, pyPrev); ctx.lineTo(px, py); ctx.stroke(); } // SPEED RAMP if (speed < maxSpeed) { speed += 0.15; } // TRIGGER BLUE FLASH if (speed > 20 && !flashTriggered) { document.getElementById("flash").style.animation = "flashAnim 1s ease-out forwards"; flashTriggered = true; } requestAnimationFrame(animate); } animate(); const btn = document.getElementById('fullscreenBtn'); if (btn) btn.addEventListener('click', () => document.fullscreenElement ? document.exitFullscreen() : document.documentElement.requestFullscreen() );
